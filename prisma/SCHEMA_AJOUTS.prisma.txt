// Préférences de l'utilisateur (confidentialité, notifications, mode vacances, adresse par défaut).
model UserPreferences {
  userId    String   @id
  data      Json
  updatedAt DateTime @updatedAt
}

// Utilisateurs bloqués.
model BlockedUser {
  id        String   @id @default(cuid())
  blockerId String
  blockedId String
  createdAt DateTime @default(now())

  @@unique([blockerId, blockedId])
  @@index([blockedId])
}

// Données de livraison d'une commande (fenêtre « Me faire livrer »).
// orderId pointe vers Order.id (volontairement sans @relation : aucune modification du model Order).
model OrderDelivery {
  id            String   @id @default(cuid())
  orderId       String   @unique
  recipientName String
  phone         String
  city          String   @default("")
  commune       String   @default("")
  quarter       String   @default("")
  address       String   @default("")
  landmark      String?
  fulfillment   String   @default("DELIVERY")      // DELIVERY | PICKUP
  paymentMethod String   @default("MOBILE_MONEY")  // WALLET | MOBILE_MONEY
  note          String?
  createdAt     DateTime @default(now())
}
